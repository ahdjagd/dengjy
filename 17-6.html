// 每一步都是一个返回 Promise 的函数
function step(name, sec) {
  return new Promise(resolve => {
    setTimeout(() => resolve(name + " 完成"), sec * 1000);
  });
}

async function makeDumplings() {
  await step("买原料", 6);      // 串行
  await step("取原料", 2);      // 串行
  await Promise.all([               // 并行！
    step("和面", 4),
    step("和馅", 5),
  ]);
  await step("包饺子", 7);
  await step("吃饺子", 3);
}
