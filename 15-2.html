if / else if / else 条件判断

function grade(score) {
    if (score >= 90) return 'A 优秀';
    else if (score >= 80) return 'B 良好';
    else if (score >= 70) return 'C 中等';
    else if (score >= 60) return 'D 及格';
    else return 'F 不及格';
}

switch 多分支选择

function weekday(day) {
    switch (day) {
        case 1: return '星期一';
        case 2: return '星期二';
        case 3: return '星期三';
        case 4: return '星期四';
        case 5: return '星期五';
        default: return '周末啦 🎉';
    }
}

for 循环 计数循环

let sum = 0;
for (let i = 1; i <= 100; i++) {
    sum += i;
}
// 累加 1+2+...+100

while / do-while 条件循环

// while：先判断后执行
let n = 1, fact = 1;
while (n <= 10) {
    fact *= n; n++;
}

// do-while：至少执行一次
let count = 0;
do {
    count++;
} while (count < 5);

嵌套 for 循环 九九乘法表

for (let i = 1; i <= 9; i++) {
    for (let j = 1; j <= i; j++) {
        process.stdout.write(
            j + '×' + i + '=' + (i*j) + '\t'
        );
    }
    console.log();
}
